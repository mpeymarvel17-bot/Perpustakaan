        <input type="text" id="author" placeholder="Penulis" required>

        <button type="submit">Simpan Buku</button>
    </form>

    <h2>Daftar Buku</h2>
    <div id="bookList">Memuat data buku...</div>

    <script>
        async function fetchBooks() {
            try {
                const res = await fetch('/api/books');
                const result = await res.json();
                const listDiv = document.getElementById('bookList');
                
                if (result.success && result.data.length > 0) {
                    listDiv.innerHTML = result.data.map(book => `
                        <div class="card">
                            <h3>${book.title}</h3>
                            <p><strong>Penulis:</strong> ${book.author} | <strong>Tahun:</strong> ${book.year}</p>
                        </div>
                    `).join('');
                } else {
                    listDiv.innerHTML = '<p>Belum ada buku di perpustakaan.</p>';
                }
            } catch (err) {
                document.getElementById('bookList').innerHTML = '<p>Gagal memuat data buku.</p>';
            }
        }

        document.getElementById('bookForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const title = document.getElementById('title').value;
            const author = document.getElementById('author').value;
            const year = document.getElementById('year').value;

            const res = await fetch('/api/books', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ title, author, year })
            });

            const result = await res.json();
            if (result.success) {
                alert('Buku berhasil ditambahkan!');
                document.getElementById('bookForm').reset();
                fetchBooks();
            } else {
                alert('Gagal: ' + result.error);
            }
        });

        fetchBooks();
    </script>
